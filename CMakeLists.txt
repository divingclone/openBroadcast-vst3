cmake_minimum_required(VERSION 3.24)
project(OpenBroadcastVST VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "A 64-bit build is required.")
endif()
set(JUCE_DIR "${CMAKE_SOURCE_DIR}/.deps/JUCE" CACHE PATH "JUCE source")
set(ORT_DIR "${CMAKE_SOURCE_DIR}/.deps/ort" CACHE PATH "Extracted ONNX Runtime SDK")
if(WIN32)
  set(ORT_INCLUDE_DIR "${ORT_DIR}/build/native/include")
  set(ORT_RUNTIME_DIR "${ORT_DIR}/runtimes/win-x64/native")
else()
  set(ORT_INCLUDE_DIR "${ORT_DIR}/include")
  set(ORT_RUNTIME_DIR "${ORT_DIR}/lib")
endif()
set(KISSFFT_DIR "${CMAKE_SOURCE_DIR}/.deps/kissfft" CACHE PATH "KissFFT source")
add_subdirectory(${JUCE_DIR} juce)
add_library(ob_engine STATIC src/Engine.cpp src/StudioVoice.cpp src/GpuDevices.cpp ${KISSFFT_DIR}/kiss_fft.c ${KISSFFT_DIR}/kiss_fftr.c)
if(WIN32)
  target_sources(ob_engine PRIVATE src/vendor/explicit_dml.cpp)
  target_include_directories(ob_engine PRIVATE "${CMAKE_SOURCE_DIR}/.deps/directml/include")
  target_link_libraries(ob_engine PUBLIC dxgi d3d12)
  target_compile_definitions(ob_engine PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_link_libraries(ob_engine PUBLIC ${CMAKE_DL_LIBS})
endif()
target_include_directories(ob_engine PUBLIC src ${ORT_INCLUDE_DIR} ${KISSFFT_DIR})
target_compile_definitions(ob_engine PUBLIC ORT_API_MANUAL_INIT)
add_library(ob_processing STATIC src/ProcessingEngine.cpp)
target_link_libraries(ob_processing PUBLIC ob_engine)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
  target_compile_definitions(ob_processing PRIVATE _GNU_SOURCE)
  add_executable(ob_worker_cuda src/LinuxWorker.cpp)
  target_link_libraries(ob_worker_cuda PRIVATE ob_engine)
  set_target_properties(ob_worker_cuda PROPERTIES OUTPUT_NAME ob_inference_worker RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/workers/cuda")
  set(CUDA_WORKER "$<TARGET_FILE:ob_worker_cuda>")
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64)$")
    set(AMD_ORT_DIR "${CMAKE_SOURCE_DIR}/.deps/onnxruntime-linux-x64-migraphx-1.27.1" CACHE PATH "MIGraphX ONNX Runtime SDK")
    if(NOT EXISTS "${AMD_ORT_DIR}/lib/libonnxruntime.so")
      message(FATAL_ERROR "MIGraphX runtime missing; run scripts/build.py to prepare both Linux runtimes.")
    endif()
    add_library(ob_engine_amd STATIC src/Engine.cpp src/StudioVoice.cpp src/GpuDevices.cpp ${KISSFFT_DIR}/kiss_fft.c ${KISSFFT_DIR}/kiss_fftr.c)
    target_include_directories(ob_engine_amd PUBLIC src "${AMD_ORT_DIR}/include" ${KISSFFT_DIR})
    target_compile_definitions(ob_engine_amd PUBLIC ORT_API_MANUAL_INIT OB_LINUX_MIGRAPHX=1)
    target_link_libraries(ob_engine_amd PUBLIC ${CMAKE_DL_LIBS})
    add_executable(ob_worker_migraphx src/LinuxWorker.cpp)
    target_link_libraries(ob_worker_migraphx PRIVATE ob_engine_amd)
    set_target_properties(ob_worker_migraphx PROPERTIES OUTPUT_NAME ob_inference_worker RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/workers/migraphx")
    set(AMD_WORKER "$<TARGET_FILE:ob_worker_migraphx>")
  endif()
endif()
juce_add_plugin(OpenBroadcast COMPANY_NAME "OpenBroadcast" PLUGIN_MANUFACTURER_CODE Obca
  PLUGIN_CODE Obv1 FORMATS VST3 PRODUCT_NAME "OpenBroadcast" IS_SYNTH FALSE
  NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE IS_MIDI_EFFECT FALSE
  COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Restoration)
target_sources(OpenBroadcast PRIVATE src/Processor.cpp src/Processor.h src/Transport.h)
target_compile_definitions(OpenBroadcast PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
  JUCE_VST3_CAN_REPLACE_VST2=0)
target_link_libraries(OpenBroadcast PRIVATE ob_processing juce::juce_audio_utils
  PUBLIC juce::juce_recommended_config_flags)
if(TARGET ob_worker_cuda)
  add_dependencies(OpenBroadcast_VST3 ob_worker_cuda)
endif()
if(TARGET ob_worker_migraphx)
  add_dependencies(OpenBroadcast_VST3 ob_worker_migraphx)
endif()
set(RESOURCE_DIR "$<TARGET_FILE_DIR:OpenBroadcast_VST3>/../Resources")
add_custom_command(TARGET OpenBroadcast_VST3 POST_BUILD
  COMMAND ${CMAKE_COMMAND} "-DDESTINATION=${RESOURCE_DIR}" "-DASSETS=${CMAKE_SOURCE_DIR}/assets"
    "-DRUNTIME=${ORT_RUNTIME_DIR}" "-DORT_SDK=${ORT_DIR}" "-DPLATFORM=${CMAKE_SYSTEM_NAME}"
    "-DCUDA_WORKER=${CUDA_WORKER}" "-DAMD_WORKER=${AMD_WORKER}" "-DAMD_ORT_SDK=${AMD_ORT_DIR}"
    "-DDIRECTML=${CMAKE_SOURCE_DIR}/.deps/directml/bin/x64-win/DirectML.dll"
    -P "${CMAKE_SOURCE_DIR}/cmake/BundleResources.cmake")
enable_testing()
add_executable(engine_test tests/engine_test.cpp)
target_link_libraries(engine_test PRIVATE ob_engine)
add_test(NAME engine COMMAND engine_test "${CMAKE_SOURCE_DIR}/assets" "${ORT_RUNTIME_DIR}" "${CMAKE_SOURCE_DIR}/tests/generated")
add_executable(studio_test tests/studio_test.cpp)
target_link_libraries(studio_test PRIVATE ob_engine)
add_test(NAME studio_cpu COMMAND studio_test "${CMAKE_SOURCE_DIR}/assets" "${ORT_RUNTIME_DIR}" "${CMAKE_SOURCE_DIR}/tests/generated")
add_executable(plugin_test tests/plugin_test.cpp)
target_link_libraries(plugin_test PRIVATE OpenBroadcast ob_engine juce::juce_audio_utils juce::juce_recommended_config_flags)
add_dependencies(plugin_test OpenBroadcast_VST3)
add_test(NAME plugin COMMAND plugin_test "${RESOURCE_DIR}")
add_executable(host_smoke tests/host_smoke.cpp)
target_link_libraries(host_smoke PRIVATE juce::juce_audio_utils juce::juce_recommended_config_flags)
target_compile_definitions(host_smoke PRIVATE JUCE_PLUGINHOST_VST3=1 JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
add_dependencies(host_smoke OpenBroadcast_VST3)
add_test(NAME vst3_bundle COMMAND host_smoke "${CMAKE_BINARY_DIR}/OpenBroadcast_artefacts/$<CONFIG>/VST3/OpenBroadcast.vst3")
add_test(NAME studio_bundle COMMAND host_smoke "${CMAKE_BINARY_DIR}/OpenBroadcast_artefacts/$<CONFIG>/VST3/OpenBroadcast.vst3" --studio)
if(TARGET ob_worker_migraphx)
  find_package(Python3 3.12 REQUIRED COMPONENTS Interpreter)
  add_library(fake_hip SHARED tests/fake_hip.cpp)
  set_target_properties(fake_hip PROPERTIES OUTPUT_NAME amdhip64 SOVERSION 7
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-hip")
  add_library(fake_cuda SHARED tests/fake_cuda.cpp)
  set_target_properties(fake_cuda PROPERTIES OUTPUT_NAME cuda SOVERSION 1
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/test-hip")
  add_executable(amd_devices_test tests/amd_devices_test.cpp src/GpuDevices.cpp)
  target_include_directories(amd_devices_test PRIVATE src)
  target_compile_definitions(amd_devices_test PRIVATE OB_LINUX_MIGRAPHX=1)
  target_link_libraries(amd_devices_test PRIVATE ${CMAKE_DL_LIBS})
  add_dependencies(amd_devices_test fake_hip)
  add_test(NAME amd_devices COMMAND ${CMAKE_COMMAND} -E env
    "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:fake_hip>" $<TARGET_FILE:amd_devices_test>)
  add_executable(amd_assignment_test tests/amd_assignment_test.cpp)
  target_link_libraries(amd_assignment_test PRIVATE ob_engine_amd)
  add_test(NAME amd_assignment COMMAND amd_assignment_test "${CMAKE_SOURCE_DIR}/assets" "${AMD_ORT_DIR}/lib")
  add_test(NAME amd_fallback COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/tests/amd_fallback_test.py"
    $<TARGET_FILE:plugin_test> "${RESOURCE_DIR}" $<TARGET_FILE_DIR:fake_hip>)
  add_executable(unified_test tests/unified_test.cpp)
  target_link_libraries(unified_test PRIVATE ob_processing)
  add_dependencies(unified_test OpenBroadcast_VST3 fake_hip fake_cuda)
  add_test(NAME unified_runtime COMMAND ${CMAKE_COMMAND} -E env
    "LD_LIBRARY_PATH=$<TARGET_FILE_DIR:fake_hip>" $<TARGET_FILE:unified_test> "${RESOURCE_DIR}")
endif()
